	\begin{abstract}
		This paper establishes Uncertain Set Theory, a novel foundational mathematical framework built from First Principles to model uncertainty without relying on probabilistic distributions, fuzzy membership degrees, or static interval bounds. Traditional approaches such as Interval Arithmetic frequently suffer from correlation loss and catastrophic numerical noise explosion, while probabilistic models require prior knowledge of probability densities that are often unavailable in real-world high-noise environments \cite{tran2026extended}. 
		
		We address these fundamental limitations by defining Uncertain Elements and Uncertain Sets directly as collections of state-dependent uncertain entities $S = \{x_i : i \in I\}$. Through this scenario-based structural formulation, we show that every uncertain set canonically induces a Configuration Space $\mathrm{Conf}(S)$, allowing set cardinality itself to unify naturally as an uncertain number $|S| \in \mathbf{U}(\mathbb{N})$, resolving cardinality degeneration and state-collapse seamlessly. Furthermore, algebraic operations on uncertain sets inherit classical binary set relations over their configuration spaces, preserving absolute logical correlation and enabling canonical complexity reduction down to $\mathcal{O}(1)$ via Lazy Evaluation. This framework serves as the core foundation for $\mathbf{U}(\mathbb{K})$ arithmetic, scenario-space optimization, and high-dimensional uncertain topology.
		
		\vspace{1em}
		\noindent Keywords: Uncertain Set Theory, Uncertain Elements, Configuration Space, State Superposition, Uncertain Cardinality, Scenario Space, Scenario-Based Arithmetic, Correlation Loss, $\mathbf{U}(\mathbb{K})$ Spaces, First Principles.
	\end{abstract}
	
	\chapter{Uncertain Sets}
	
	\begin{definition}[Uncertain Element]
		Let $U$ be a non-empty universal set. An uncertain element $x$ on $U$ is an object taking an undetermined value in $U$, constrained by a set of possible values $A \subseteq U$ ($A \neq \emptyset$), denoted by:
		\[
		x \in A
		\]
		The set $A$ is called the possibility domain (or state space) of $x$, denoted by $\mathrm{Val}(x) = A$.
	\end{definition}
	
	\begin{definition}[Uncertain Set]
		Let $U$ be a non-empty universal set and $I$ be an index set. An uncertain set $S$ on $U$ is a collection of uncertain elements on $U$:
		\[
		S = \{x_i : i \in I\}
		\]
		where each uncertain element $x_i$ ($i \in I$) has a possibility domain $\mathrm{Val}(x_i) \subseteq U$.
	\end{definition}
	
	\begin{definition}[Configuration Space]
		Let $S = \{x_i : i \in I\}$ be an uncertain set on $U$. A realized configuration (or empirical state) of $S$ is a classical set $S^* \subseteq U$ obtained when the constituent elements $x_i$ simultaneously realize specific values $v_i \in \mathrm{Val}(x_i)$:
		\[
		S^* = \{v_i : i \in I\}
		\]
		The collection of all possible classical configurations realized by $S$ is called the configuration space of $S$, denoted by $\mathrm{Conf}(S)$.
	\end{definition}
	
	\begin{definition}[Cardinality Operator]
		Let $A$ be a classical set. The cardinality (or cardinal number) of $A$, denoted by $\mathrm{card}(A)$ or $|A|$, represents the measure of the number of elements in $A$. For any configuration $S^* \in \mathrm{Conf}(S)$, $\mathrm{card}(S^*)$ maps $S^*$ to its non-negative cardinal measure in $\mathbb{N} \cup \{\infty\}$.
	\end{definition}
	
	\begin{definition}[Uncertain Cardinality]
		Let $S = \{x_i : i \in I\}$ be an uncertain set on the universal set $U$. The uncertain cardinality (or size) of $S$, denoted by $|S|$ or $\mathrm{card}(S)$, is an uncertain number on the set of natural numbers $\mathbb{N}$ whose possibility domain is given by the cardinalities of all realized states in its configuration space $\mathrm{Conf}(S)$:
		\[
		\mathrm{Val}(|S|) = \mathrm{Val}(\mathrm{card}(S)) = \{ \mathrm{card}(S^*) : S^* \in \mathrm{Conf}(S) \} \subseteq \mathbb{N}
		\]
		where $\mathrm{card}(S^*) = |S^*|$ represents the classical cardinality of the configuration $S^*$.
	\end{definition}
	
	\begin{remark}
		If all uncertain elements $x_i$ in $S$ take distinct values across all realized configurations, then $|S^*| = |I|$ for every $S^* \in \mathrm{Conf}(S)$, in which case the uncertain cardinality degenerates into a classical constant $|S| = |I|$. Conversely, size degeneration (cardinality collapse) occurs at configurations $S^*$ where duplicate values appear among elements ($v_i = v_j$ for $i \neq j$).
	\end{remark}
		
	\begin{example}[Numerical Illustration and Uncertain Cardinality]
		Consider the universal set $U = \mathbb{R}$. Let $x_1$ and $x_2$ be two uncertain elements with possibility domains:
		\[
		\mathrm{Val}(x_1) = \{1, 2\}, \quad \mathrm{Val}(x_2) = \{1, 3\}
		\]
		Let $S = \{x_1, x_2\}$ be an uncertain set.
		
		Assuming $x_1$ and $x_2$ vary independently, the configuration space $\mathrm{Conf}(S)$ consists of $2 \times 2 = 4$ realized classical states:
		\begin{itemize}
			\item When $x_1 = 1, x_2 = 1 \implies S_1^* = \{1, 1\} = \{1\}$, so $|S_1^*| = \mathrm{card}(S_1^*) = 1$.
			\item When $x_1 = 1, x_2 = 3 \implies S_2^* = \{1, 3\}$, so $|S_2^*| = \mathrm{card}(S_2^*) = 2$.
			\item When $x_1 = 2, x_2 = 1 \implies S_3^* = \{2, 1\}$, so $|S_3^*| = \mathrm{card}(S_3^*) = 2$.
			\item When $x_1 = 2, x_2 = 3 \implies S_4^* = \{2, 3\}$, so $|S_4^*| = \mathrm{card}(S_4^*) = 2$.
		\end{itemize}
		
		Therefore, the configuration space of $S$ is explicitly:
		\[
		\mathrm{Conf}(S) = \big\{ \{1\}, \{1, 3\}, \{1, 2\}, \{2, 3\} \big\}
		\]
		Hence, the uncertain cardinality $|S|$ has a possibility domain that constitutes an uncertain number on $\mathbb{N}$:
		\[
		\mathrm{Val}(|S|) = \{ |S^*| : S^* \in \mathrm{Conf}(S) \} = \{1, 2\} \subseteq \mathbb{N}
		\]
	\end{example}
	
	\begin{definition}[Element Membership via Configuration Space Measure]
		Let $S = \{x_1, x_2, \dots, x_n\}$ be an uncertain set on a non-empty universal set $U$, where each uncertain element $x_i$ ($i \in \{1, \dots, n\}$) has an independent possibility domain $\mathrm{Val}(x_i) \subseteq U$. 
		
		The global scenario space of $S$ is defined by the Cartesian product:
		\[
		\Omega_S = \mathrm{Val}(x_1) \times \mathrm{Val}(x_2) \times \dots \times \mathrm{Val}(x_n)
		\]
		equipped with a product measure $\lambda$ on $\Omega_S$. Each state tuple $\mathbf{v} = (v_1, \dots, v_n) \in \Omega_S$ uniquely induces a realized configuration $S^*(\mathbf{v}) = \{v_1, \dots, v_n\} \in \mathrm{Conf}(S)$.
		
		For a classical element $u \in U$, let $E_u \subseteq \Omega_S$ be the measurable set of scenarios whose induced configuration contains $u$:
		\[
		E_u = \{\mathbf{v} \in \Omega_S \;:\; u \in S^*(\mathbf{v})\}
		\]
		The truth value of the membership assertion $u \in_u S$ is measured by:
		\[
		\mathcal{V}(u \in_u S) := \frac{\lambda(E_u)}{\lambda(\Omega_S)} \in [0, 1]
		\]
	\end{definition}
	
	\begin{example}[Numerical Illustration of Element Membership via Configuration Measure]
		Consider the universal set $U = \mathbb{R}$. Let $x_1, x_2$ be uncertain elements with independent possibility domains:
		\[
		\mathrm{Val}(x_1) = \{1, 3, 5, 7\}, \quad \mathrm{Val}(x_2) = \{2, 3, 4, 5\}
		\]
		Let $S = \{x_1, x_2\}$ be an uncertain set.
		
		The joint scenario space of state tuples is $\Omega_S = \mathrm{Val}(x_1) \times \mathrm{Val}(x_2)$, equipped with the counting measure $\lambda(A) = |A|$. The total measure is $\lambda(\Omega_S) = 4 \times 4 = 16$.
		
		The 16 realized state tuples $\mathbf{v} = (v_1, v_2) \in \Omega_S$ and their corresponding realized configurations $S^*(\mathbf{v}) = \{v_1, v_2\} \in \mathrm{Conf}(S)$ are listed as:
		\begin{align*}
			\Omega_S = \{ &(1,2), (1,3), (1,4), (1,5), (3,2), (3,3), \\
			&(3,4), (3,5), (5,2), (5,3), (5,4), (5,5), \\
			&(7,2), (7,3), (7,4), (7,5) \}
		\end{align*}
		
		Evaluating the truth measure of proposition $3 \in_u S$, the subset $E_3 \subseteq \Omega_S$ satisfying $3 \in S^*(\mathbf{v})$ is:
		\[
		E_3 = \{(1,3), (3,2), (3,3), (3,4), (3,5), (5,3), (7,3)\}
		\]
		Since $\lambda(E_3) = |E_3| = 7$, the truth valuation yields:
		\[
		\mathcal{V}(3 \in_u S) = \frac{\lambda(E_3)}{\lambda(\Omega_S)} = \frac{7}{16} = 0.4375
		\]
	\end{example}
	
	\begin{theorem}[Conservation of Total Truth Valuation]
		Let $U$ be a non-empty universal set and let $S = \{x_1, x_2, \dots, x_n\}$ be an uncertain set on $U$, equipped with a finite product measure $\lambda$ on its scenario space $\Omega_S$. 
		
		If $U$ is discrete (or if $\mathrm{Conf}(S)$ contains finite support over $U$), then the sum of the membership truth valuations of all elements in $U$ equals the expected value of the uncertain cardinality of $S$ over the scenario space $\Omega_S$:
		\[
		\sum_{u \in U} \mathcal{V}(u \in_u S) = \mathbb{E}\left[|S|\right]
		\]
		where $\mathbb{E}\left[|S|\right]$ is defined by:
		\[
		\mathbb{E}\left[|S|\right] = \frac{1}{\lambda(\Omega_S)} \int_{\Omega_S} |S^*(\mathbf{v})| \, d\lambda(\mathbf{v})
		\]
		and $|S^*(\mathbf{v})|$ represents the classical cardinality of the realized state $S^*(\mathbf{v}) = \{v_1, \dots, v_n\} \in \mathrm{Conf}(S)$ at configuration $\mathbf{v} = (v_1, \dots, v_n) \in \Omega_S$.
	\end{theorem}
	
	\begin{proof}
		By definition of the membership truth valuation under measure $\lambda$, for each element $u \in U$, we have:
		\[
		\mathcal{V}(u \in_u S) = \frac{\lambda(E_u)}{\lambda(\Omega_S)}
		\]
		where $E_u = \{\mathbf{v} \in \Omega_S \;:\; u \in S^*(\mathbf{v})\}$ is the measurable set of configurations containing $u$.
		
		Expressing $\lambda(E_u)$ using the indicator function $\mathbb{I}_{\{u \in S^*(\mathbf{v})\}}(\mathbf{v})$, we obtain:
		\[
		\lambda(E_u) = \int_{\Omega_S} \mathbb{I}_{\{u \in S^*(\mathbf{v})\}}(\mathbf{v}) \, d\lambda(\mathbf{v})
		\]
		
		Summing over all elements $u \in U$:
		\[
		\sum_{u \in U} \mathcal{V}(u \in_u S) = \frac{1}{\lambda(\Omega_S)} \sum_{u \in U} \int_{\Omega_S} \mathbb{I}_{\{u \in S^*(\mathbf{v})\}}(\mathbf{v}) \, d\lambda(\mathbf{v})
		\]
		
		By Tonelli's Theorem for non-negative functions, we interchange the sum and the integral:
		\begin{align*}
			\sum_{u \in U} \int_{\Omega_S} \mathbb{I}_{\{u \in S^*(\mathbf{v})\}}(\mathbf{v}) \, d\lambda(\mathbf{v}) &= \int_{\Omega_S} \left( \sum_{u \in U} \mathbb{I}_{\{u \in S^*(\mathbf{v})\}}(\mathbf{v}) \right) d\lambda(\mathbf{v})
		\end{align*}
		
		Notice that for a fixed configuration $\mathbf{v} \in \Omega_S$, the sum of indicators over all $u \in U$ simply counts the number of distinct elements in the realized set $S^*(\mathbf{v})$:
		\[
		\sum_{u \in U} \mathbb{I}_{\{u \in S^*(\mathbf{v})\}}(\mathbf{v}) = |S^*(\mathbf{v})|
		\]
		
		Substituting this identity back into the integral yields:
		\begin{align*}
			\sum_{u \in U} \mathcal{V}(u \in_u S) &= \frac{1}{\lambda(\Omega_S)} \int_{\Omega_S} |S^*(\mathbf{v})| \, d\lambda(\mathbf{v}) \\
			&= \mathbb{E}\left[|S|\right]
		\end{align*}
		This completes the proof.
	\end{proof}
	
	\begin{example}[Illustration of Total Valuation Conservation]
		Revisiting $S = \{x_1, x_2\}$ with $\mathrm{Val}(x_1) = \{1, 3, 5, 7\}$ and $\mathrm{Val}(x_2) = \{2, 3, 4, 5\}$, where $\lambda(\Omega_S) = 16$.
		
		The elements $u \in U$ with non-zero membership valuation are evaluated as follows:
		\begin{align*}
			\mathcal{V}(1 \in_u S) = \frac{4}{16}, \quad \mathcal{V}(2 \in_u S) = \frac{4}{16}, \quad \mathcal{V}(3 \in_u S) = \frac{7}{16}, \\
			\mathcal{V}(4 \in_u S) = \frac{4}{16}, \quad \mathcal{V}(5 \in_u S) = \frac{7}{16}, \quad \mathcal{V}(7 \in_u S) = \frac{4}{16}.
		\end{align*}
		
		Summing these truth valuations over the universe $U$:
		\[
		\sum_{u \in U} \mathcal{V}(u \in_u S) = \frac{4 + 4 + 7 + 4 + 7 + 4}{16} = \frac{30}{16} = 1.875
		\]
		
		On the other hand, among the 16 realized configurations in $\mathrm{Conf}(S)$:
		\begin{itemize}[leftmargin=*]
			\item 14 configurations yield distinct pairs $x_1 \neq x_2$ with cardinality $|S^*| = 2$.
			\item 2 configurations yield duplicate elements $(3,3)$ and $(5,5)$ with collapsed cardinality $|S^*| = 1$.
		\end{itemize}
		
		Computing the expected value of the uncertain cardinality $\mathbb{E}\left[|S|\right]$:
		\[
		\mathbb{E}\left[|S|\right] = \frac{14 \times 2 + 2 \times 1}{16} = \frac{30}{16} = 1.875
		\]
		Thus, $\sum_{u \in U} \mathcal{V}(u \in_u S) = \mathbb{E}\left[|S|\right]$ holds identically.
	\end{example}
